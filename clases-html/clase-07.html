Conexión WiFi (Modo STA)

Código Básico

#include <WiFi.h>

const char* ssid = "NombreDeTuRed";
const char* password = "Contrasena";

void setup() {
    Serial.begin(115200);
    WiFi.begin(ssid, password);
    
    Serial.print("Conectando");
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    
    Serial.println("\nConectado!");
    Serial.print("IP: ");
    Serial.println(WiFi.localIP());
}

void loop() {}

Servidor Web en ESP32

Código Básico

#include <WiFi.h>
#include <WebServer.h>

const char* ssid = "TU_RED";
const char* password = "TU_PASS";

WebServer server(80);  // Puerto 80 (HTTP)

void handleRoot() {
    String html = "";
    html += "
Hola desde ESP32!
";
    html += "
Temperatura: 25°C
";
    html += "";
    server.send(200, "text/html", html);
}

void setup() {
    Serial.begin(115200);
    WiFi.begin(ssid, password);
    while (WiFi.status() != WL_CONNECTED) delay(500);
    
    Serial.println(WiFi.localIP());
    
    server.on("/", handleRoot);
    server.begin();
}

void loop() {
    server.handleClient();
}
Acceder: En navegador, ir a la IP mostrada en el Serial Monitor (ej: 192.168.1.100)

ThingSpeak: Plataforma IoT Gratuita

¿Qué es ThingSpeak?

  • Almacenar datos de sensores en la nube
  • Visualizar en gráficos automáticos
  • API simple basada en HTTP
  • Gratuito (con limitaciones)

Configuración

  1. Crear cuenta en thingspeak.com
  2. Crear nuevo "Channel"
  3. Configurar Field 1: Temperatura, Field 2: Humedad
  4. Obtener Write API Key

Código para ThingSpeak

#include <WiFi.h>
#include <HTTPClient.h>

const char* apiKey = "TU_API_KEY";

void enviarAThingSpeak(float temp, float hum) {
    if (WiFi.status() != WL_CONNECTED) return;
    
    HTTPClient http;
    String url = "http://api.thingspeak.com/update?api_key=";
    url += apiKey;
    url += "&field1=" + String(temp);
    url += "&field2=" + String(hum);
    
    http.begin(url);
    int httpCode = http.GET();
    
    if (httpCode == 200) {
        Serial.println("Datos enviados!");
    }
    http.end();
}

// En loop(), llamar cada 15 segundos mínimo:
// enviarAThingSpeak(temperatura, humedad);
