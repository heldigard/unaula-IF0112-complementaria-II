💻 Estructura de Código Arduino

// ===========================================
// VARIABLES GLOBALES (arriba del setup)
// ===========================================
const int PIN_LED = 2;
const int PIN_BOTON = 4;
int contador = 0;

// ===========================================
// SETUP - Se ejecuta UNA VEZ al inicio
// ===========================================
void setup() {
  Serial.begin(115200);        // Iniciar comunicación serial

  pinMode(PIN_BOTON, INPUT);   // Configurar pin como entrada

}

// ===========================================
// LOOP - Se ejecuta repetidamente
// ===========================================
void loop() {
  // Tu código aquí
  delay(1000);  // Esperar 1 segundo
}

🔌 Códigos Básicos

LED Parpadear (Blink)

const int LED = 2;

void setup() {
  pinMode(LED, OUTPUT);
}

void loop() {
  digitalWrite(LED, HIGH);  // Encender
  delay(1000);              // 1 segundo
  digitalWrite(LED, LOW);   // Apagar
  delay(1000);              // 1 segundo
}

Leer Botón

const int BOTON = 4;
const int LED = 2;

void setup() {
  pinMode(BOTON, INPUT_PULLUP);  // Pull-up interno
  pinMode(LED, OUTPUT);
}

void loop() {
  if (digitalRead(BOTON) == LOW) {
    digitalWrite(LED, HIGH);  // Presionado = LOW
  } else {
    digitalWrite(LED, LOW);
  }
}

Leer Sensor DHT22

#include "DHT.h"

#define DHTPIN 4
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(115200);
  dht.begin();
}

void loop() {
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();

  Serial.print("Temp: ");
  Serial.print(temp);
  Serial.print("°C  Hum: ");
  Serial.print(hum);
  Serial.println("%");

  delay(3000);  // Esperar 3 segundos
}

Leer LDR (Analógico)

#define LDR_PIN 34

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);  // ESP32: 12 bits = 0-4095
}

void loop() {
  int valor = analogRead(LDR_PIN);

  Serial.print("Luz: ");
  Serial.println(valor);

  delay(500);
}

const char* servidor = "http://192.168.1.100:5000/api/v1/datos";

void enviarDatos(float temp, float hum, int luz) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;

    // Crear JSON
    StaticJsonDocument<256> doc;
    doc["temperatura"] = temp;
    doc["humedad"] = hum;
    doc["luz"] = luz;

    String jsonString;
    serializeJson(doc, jsonString);

    // Enviar
    http.begin(servidor);
    http.addHeader("Content-Type", "application/json");
    int httpCode = http.POST(jsonString);

    if (httpCode > 0) {
      Serial.println("Enviado OK");
    }

    http.end();
  }
}

🐛 Debugging Rápido

¿Algo no funciona?

1. Verificar conexiones

  • ¿Está conectado al pin correcto?
  • ¿GND conectado?
  • ¿3.3V o 5V conectado?

2. Usar Monitor Serial

Serial.begin(115200);
Serial.println("Debug: paso 1");

3. Verificar valores

Serial.print("Valor: ");
Serial.println(variable);

4. Dividir y conquistar

  • Comentar partes del código
  • Probar cada parte por separado
  • Volver agregando gradualmente

⚠️ Errores Comunes

Error Solución
"'DHT' was not declared" Falta incluir librería: #include "DHT.h" o instalar librería en Arduino IDE
"Compilation error" Verificar punto y coma, paréntesis balanceados {}, comillas en strings
"Port not found" ESP32 no conectado, drivers no instalados, puerto incorrecto
"Failed to connect" Presionar botón BOOT mientras sube, verificar puerto, cambiar cable USB

📊 Conversiones

ADC (Entrada Analógica)

ESP32 ADC: 12 bits = 0 a 4095

// Convertir a porcentaje:
porcentaje = map(valorLDR, 0, 4095, 0, 100);

// Convertir a voltaje (3.3V):
voltaje = (valor * 3.3) / 4095;

Temperatura

// Celsius a Fahrenheit:
F = (C * 9/5) + 32;

// Fahrenheit a Celsius:
C = (F - 32) * 5/9;

🚨 Modo Emergencia

Código no sube

  1. Presionar BOOT mientras sube
  2. Presionar RESET antes de subir
  3. Verificar puerto correcto
  4. Simplificar código (comentar partes)

ESP32 caliente

  • Normal durante transmisión WiFi
  • Preocupante si > 60°C
  • Verificar cortocircuito en GPIO
